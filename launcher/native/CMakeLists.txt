cmake_minimum_required(VERSION 3.22)
project(ReawakenedNativeLauncher LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "Windows required")
endif()
add_executable(ReawakenedLauncher WIN32 ReawakenedLauncher.cpp)
target_compile_features(ReawakenedLauncher PRIVATE cxx_std_17)
target_compile_definitions(ReawakenedLauncher PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(ReawakenedLauncher PRIVATE comctl32 crypt32 shell32 advapi32 user32 gdi32)
if(MSVC)
    target_compile_options(ReawakenedLauncher PRIVATE /W4 /EHsc /permissive- /utf-8)
    target_link_options(ReawakenedLauncher PRIVATE /DYNAMICBASE /NXCOMPAT)
endif()
