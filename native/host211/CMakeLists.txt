cmake_minimum_required(VERSION 3.20)
project(ReawakenedHost211 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../tools/generate_host211_profile.py" "${CMAKE_CURRENT_SOURCE_DIR}/profile.json" "${CMAKE_CURRENT_BINARY_DIR}/profile_generated.hpp" RESULT_VARIABLE generated)
if(NOT generated EQUAL 0)
  message(FATAL_ERROR "Could not generate native profile")
endif()
add_library(host211 STATIC bindings.cpp world_attachment.cpp)
target_include_directories(host211 PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
if(WIN32)
  target_link_libraries(host211 PRIVATE bcrypt)
endif()
if(MSVC)
  target_compile_options(host211 PRIVATE /W4 /WX)
else()
  target_compile_options(host211 PRIVATE -Wall -Wextra -Werror)
endif()
add_executable(host211_tests test_host.cpp)
target_link_libraries(host211_tests PRIVATE host211)
add_executable(host211_world_tests test_world_attachment.cpp)
target_link_libraries(host211_world_tests PRIVATE host211)
enable_testing()
add_test(NAME host_initialization_gates COMMAND host211_tests)
add_test(NAME world_driver_handoff COMMAND host211_world_tests)
